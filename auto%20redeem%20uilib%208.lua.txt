-- ============================================================
--  fn52 do Von: checa se msg bate com algum trigger
--  retorna: matched(bool), triggerPhrase(str|nil), textAfter(str|nil)
-- ============================================================
local function checkTrigger(text)
    if not text or text=="" then return false end

    -- se todos triggers são vazios → match-all (qualquer mensagem)
    local hasAny = false
    for i=1,10 do if triggers[i]~="" then hasAny=true; break end end
    if not hasAny then return true, nil, text end

    local lower = text:lower()
    for i=1,10 do
        local t = triggers[i]
        if t~="" then
            local s, e = lower:find(t:lower(), 1, true)
            if e then
                local after = text:sub(e+1):match("^%s*(.*)")
                return true, t, after
            end
        end
    end
    return false
end

-- ============================================================
--  sanitise (igual ao Von fn61): remove non-alnum, cap 50
-- ============================================================
local function sanitise(s)
    return string.sub(string.gsub(tostring(s), "[^%w]", ""), 1, 50)
end

-- ============================================================
--  fn56 do Von: acha TextBox e botão Confirm
--  caminho exato: PlayerGui.Codes → Codes → CodeRedeem → TextBox
--                 PlayerGui.Codes → Codes → Confirm → (TextButton filho)
-- ============================================================
local _codeBox = nil   -- v89 no Von
local _confirmBtn = nil  -- v90 no Von

local function findCodeUI(timeout)
    -- cache
    if _codeBox and _codeBox.Parent and _confirmBtn and _confirmBtn.Parent then
        return _codeBox, _confirmBtn
    end
    _codeBox = nil; _confirmBtn = nil; submitSignal = nil

    local codesGui = findCodesGui(timeout)
    if not codesGui then error("PlayerGui.Codes not found") end

    -- caminho exato do Von: Codes → CodeRedeem → TextBox
    local inner = codesGui:FindFirstChild("Codes")
    if inner then
        local codeRedeem = inner:FindFirstChild("CodeRedeem")
        if codeRedeem then
            _codeBox = codeRedeem:FindFirstChild("TextBox") or codeRedeem:FindFirstChildWhichIsA("TextBox")
        end
        local confirm = inner:FindFirstChild("Confirm")
        if confirm then
            _confirmBtn = confirm:FindFirstChildWhichIsA("TextButton")
                or confirm:FindFirstChildWhichIsA("ImageButton")
                or confirm
        end
    end

    -- fallback genérico se o caminho exato não existir
    if not _codeBox then
        for _, d in ipairs(codesGui:GetDescendants()) do
            local n = d.Name:lower()
            if d:IsA("TextBox") and (n:find("code") or n:find("input") or n:find("redeem")) then
                _codeBox = d; break
            end
        end
        if not _codeBox then
            for _, d in ipairs(codesGui:GetDescendants()) do
                if d:IsA("TextBox") then _codeBox = d; break end
            end
        end
    end
    if not _confirmBtn then
        for _, d in ipairs(codesGui:GetDescendants()) do
            if d:IsA("TextButton") or d:IsA("ImageButton") then
                local n=d.Name:lower(); local t=""
                pcall(function() t=d.Text:lower() end)
                if n:find("confirm") or n:find("submit") or n:find("redeem")
                or t:find("confirm") or t:find("submit") or t:find("redeem") then
                    _confirmBtn = d; break
                end
            end
        end
    end

    if not _codeBox    then error("TextBox de código não encontrado") end
    if not _confirmBtn then error("botão Confirm não encontrado") end
    return _codeBox, _confirmBtn
end

-- ============================================================
--  fn59 do Von: firesignal no botão (detecta MouseButton1Click ou Activated)
-- ============================================================
local function fireBtn(btn)
    if typeof(firesignal) ~= "function" then
        error("executor sem firesignal")
    end
    -- usa cache do sinal se já detectado
    if submitSignal == "click" then
        firesignal(btn.MouseButton1Click); return "MouseButton1Click"
    end
    if submitSignal == "activated" then
        firesignal(btn.Activated); return "Activated"
    end
    -- detecta qual sinal tem listeners
    if typeof(getconnections) == "function" then
        local ok, cs = pcall(getconnections, btn.MouseButton1Click)
        if ok and cs and #cs > 0 then
            submitSignal = "click"
            firesignal(btn.MouseButton1Click); return "MouseButton1Click"
        end
        local ok2, cs2 = pcall(getconnections, btn.Activated)
        if ok2 and cs2 and #cs2 > 0 then
            submitSignal = "activated"
            firesignal(btn.Activated); return "Activated"
        end
    end
    -- sem detecção: tenta MouseButton1Click mesmo assim
    firesignal(btn.MouseButton1Click); return "MouseButton1Click (unverified)"
end

-- ============================================================
--  fn57 do Von: escreve código no TextBox e dispara botão
-- ============================================================
local function submitViaFiresignal(code)
    -- tenta achar UI sem timeout primeiro (cache)
    local ok = pcall(findCodeUI, 0)
    if not ok or not _codeBox then
        -- UI fechada: abre e aguarda
        openCodesUI()
        task.wait(0.5)
        pcall(findCodeUI, 5)  -- espera até 5s pela UI aparecer
    end

    if not _codeBox    then error("TextBox não encontrado") end
    if not _confirmBtn then error("botão Confirm não encontrado") end

    -- Von fn57: box.Text = code, depois firesignal no botão
    _codeBox.Text = code
    RunService.Heartbeat:Wait()

    return fireBtn(_confirmBtn)
end

-- ============================================================
--  fn58 do Von: submit via Remote (FireServer / InvokeServer)
-- ============================================================
local function submitViaRemote(code)
    if not codeRemote then error("code remote não capturado") end
    if isRemoteEvt then
        codeRemote:FireServer(code)
        return "FireServer"
    else
        local ok, result = pcall(codeRemote.InvokeServer, codeRemote, code)
        if not ok then error(result) end
        return "InvokeServer → "..tostring(result)
    end
end

-- ============================================================
--  fn60 do Von: submit principal
--  Von tenta: remote (se premium) → firesignal
--  Nosso: remote (se capturado) → firesignal
-- ============================================================
local function doRedeem(raw)
    local code = sanitise(raw)
    if code == "" then
        addLog("nada após sanitise — não enviado", COL_OFF); return false
    end

    -- debounce 2s (igual ao Von)
    local now = os.clock()
    if code == lastCode and now - lastCodeAt < 2 then
        addLog("["..code.."] já enviado há "..math.floor((now-lastCodeAt)*1000).."ms — ignorado", COL_YEL)
        return false
    end
    lastCode   = code
    lastCodeAt = now

    if not _autoSubmit then
        setStatus("Captured: "..code, COL_ACC)
        addLog("captured (auto-submit off): "..code, COL_ACC)
        lib.ShowNotification("Captured", code)
        return true
    end

    setStatus("Submitting: "..code, COL_YEL)
    addLog("submit → "..code, COL_YEL)

    local ok, result

    -- Tenta remote se capturado
    if codeRemote then
        ok, result = pcall(submitViaRemote, code)
        if not ok then
            addLog("remote falhou ("..tostring(result)..") — tentando firesignal", COL_YEL)
        end
    end

    -- Fallback: firesignal no botão (método principal do Von sem premium)
    if not ok then
        if typeof(firesignal) == "function" then
            ok, result = pcall(submitViaFiresignal, code)
        else
            ok, result = false, "executor sem firesignal e sem remote capturado"
        end
    end

    if ok then
        setStatus("Sent ✓ "..code, COL_ON)
        addLog("sent ✓ via "..tostring(result).." → "..code, COL_ON)
        lib.ShowNotification("Sent ✓", code)
    else
        lastCode = nil  -- permite tentar de novo
        setStatus("FAIL: "..tostring(result), COL_OFF)
        addLog("FAIL ("..tostring(result).."): "..code, COL_OFF)
    end
    return ok
end

-- ============================================================
--  fn59 do Von: processa cada notificação recebida
-- ============================================================
local function onNotification(text)
    if not _enabled then return end
    if not text or text=="" then return end

    -- strip rich text
    text = text:gsub("<[^>]->",""):match("^%s*(.-)%s*$") or ""
    if text=="" then return end

    -- dedup (igual ao Von: 250ms)
    local now = os.clock()
    if text==lastText and now-lastTextAt < 0.25 then
        addLog("duplicada em "..math.floor((now-lastTextAt)*1000).."ms - ignorada", lib.Theme.TextMuted)
        return
    end
    lastText   = text
    lastTextAt = now

    -- ── Fase 2: coletar partes do código (já em coleta) ──
    if collecting then
        collected += 1
        parts[collected] = text
        addLog("parte #"..collected.."/"..totalNeeded..": "..text, COL_YEL)

        if collected >= totalNeeded then
            local code = table.concat(parts)
            resetCollect()
            doRedeem(code)
        end
        return
    end

    -- ── Fase 1: checar trigger ──
    local matched, phrase, after = checkTrigger(text)

    if not matched then
        addLog("sem match: "..text, lib.Theme.TextMuted)
        return
    end

    addLog("trigger match"..(phrase and " \""..phrase.."\"" or "")..": "..text, COL_YEL)

    -- se o código já vem inline após o trigger na mesma mensagem
    if after and #after > 0 then
        local code = sanitise(after)
        if #code > 0 then
            addLog("inline code: "..code, COL_ON)
            resetCollect()
            doRedeem(code)
            return
        end
    end

    -- senão: iniciar coleta de N notificações seguintes
    resetCollect()
    collecting  = true
    totalNeeded = math.max(WORD_COUNT, 1)
    addLog("coletando "..totalNeeded.." notificação(ões) seguintes...", COL_YEL)
    setStatus("Aguardando "..totalNeeded.." parte(s)...", COL_YEL)
end

-- ============================================================
--  Resolve notif remote (para listener e teste)
-- ============================================================
local function resolveNotifRemote()
    if _G.PhiNotifyRemote then return _G.PhiNotifyRemote end
    local okP, Packages = pcall(function() return ReplicatedStorage:WaitForChild("Packages",8) end)
    if okP and Packages then
        local okN, Net = pcall(function() return Packages:WaitForChild("Net",5) end)
        if okN and Net then
            local gi = debug and (debug.getinfo or debug.info)
            if getgc and gi and getconnections then
                for _, d in ipairs(Net:GetDescendants()) do
                    if d:IsA("RemoteEvent") then
                        local ok, cs = pcall(getconnections, d.OnClientEvent)
                        if ok then
                            for _, c in ipairs(cs) do
                                local f, fn = pcall(function() return c.Function end)
                                if f and type(fn)=="function" then
                                    local i, info = pcall(gi, fn)
                                    if i and tostring(info.short_src or info.source or ""):find("NotificationController",1,true) then
                                        return d
                                    end
                                end
                            end
                        end
                    end
                end
            end
            for _, d in ipairs(Net:GetDescendants()) do
                if d:IsA("RemoteEvent") then
                    local ln=d.Name:lower()
                    if ln:find("notif") or ln:find("announce") or ln:find("message") then return d end
                end
            end
        end
    end
    for _, d in ipairs(ReplicatedStorage:GetDescendants()) do
        if d:IsA("RemoteEvent") then
            local ln=d.Name:lower()
            if ln:find("notif") or ln:find("announce") then return d end
        end
    end
    return nil
end

-- ============================================================
--  Busca remote de código (Von tbl12: namecall hook / scan)
--  Método simplificado: escaneia RemoteFunction/RemoteEvent em Codes GUI
-- ============================================================
local function findCodeRemote()
    local codesGui = PlayerGui:FindFirstChild("Codes")
    if not codesGui then
        openCodesUI(); task.wait(0.5)
        codesGui = PlayerGui:FindFirstChild("Codes")
    end
    if not codesGui then return nil, false end

    -- busca RemoteFunction/RemoteEvent descendente de Codes
    for _, d in ipairs(codesGui:GetDescendants()) do
        if d:IsA("RemoteFunction") then return d, false end
        if d:IsA("RemoteEvent") then return d, true end
    end
    -- busca em ReplicatedStorage por nome relacionado a código
    for _, d in ipairs(ReplicatedStorage:GetDescendants()) do
        if d:IsA("RemoteFunction") or d:IsA("RemoteEvent") then
            local n = d.Name:lower()
            if n:find("code") or n:find("redeem") or n:find("claim") then
                return d, d:IsA("RemoteEvent")
            end
        end
    end
    return nil, false
end

-- ============================================================
--  Listener: notifRemote.OnClientEvent  →  onNotification
-- ============================================================
local listenConn
task.spawn(function()
    notifRemote = resolveNotifRemote()
    if notifRemote then
        addLog("notif remote: "..notifRemote:GetFullName(), COL_ON)
        if getgenv then
            local prev = getgenv().CyberCodeSniperNotifyConnection
            if prev then pcall(function() prev:Disconnect() end) end
        end
        listenConn = notifRemote.OnClientEvent:Connect(function(...)
            -- Von passa: text, duration, sound, position, userId
            local text = tostring((...) or "")
            pcall(onNotification, text)
        end)
        if getgenv then getgenv().CyberCodeSniperNotifyConnection = listenConn end
        addLog("listening ✓", COL_ON)
    else
        addLog("notif remote NOT found", COL_OFF)
    end

    -- tenta capturar remote de código em background
    task.spawn(function()
        local r, isEvt = findCodeRemote()
        if r then
            codeRemote  = r
            isRemoteEvt = isEvt
            addLog("code remote: "..r:GetFullName().." (".. (isEvt and "RemoteEvent" or "RemoteFunction") ..")", COL_ON)
        else
            addLog("code remote não encontrado (usará firesignal)", COL_YEL)
        end
    end)
end)

-- ============================================================
--  watchLabel: monitora TextLabels do PlayerGui
--  (captura fallback e rejeições do jogo)
-- ============================================================
local watchedLabels = {}
local function watchLabel(obj)
    if not (obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox")) then return end
    -- ignora nossa própria GUI
    local p=obj; for _=1,12 do
        if not p then break end
        if p.Name=="HighRedeem" or p.Name=="_HighTestNotif" then return end
        p=p.Parent
    end
    if watchedLabels[obj] then return end
    watchedLabels[obj]=true

    obj:GetPropertyChangedSignal("Text"):Connect(function()
        local t = obj.Text or ""; if t=="" then return end
        local lo = t:lower()
        -- rejeições do jogo
        if lo:find("invalid code") or lo:find("expired") or lo:find("already redeemed")
        or lo:find("already used") or lo:find("doesn't exist") or lo:find("does not exist")
        or lo:find("not found") or lo:find("unexpected error") then
            setStatus("Game: "..t, COL_OFF); addLog("game reject: "..t, COL_OFF)
            resetCollect(); return
        end
        -- dispara como notificação (fallback do GUI test)
        pcall(onNotification, t)
    end)
end
for _, o in ipairs(PlayerGui:GetDescendants()) do watchLabel(o) end
PlayerGui.DescendantAdded:Connect(function(o) task.defer(function() watchLabel(o) end) end)

-- ============================================================
--  FPS/Ping loop
-- ============================================================
local frames,elapsed=0,0
RunService.RenderStepped:Connect(function(dt)
    frames+=1; elapsed+=dt
    if elapsed>=0.5 then
        local fps=math.clamp(math.floor(frames/elapsed+0.5),0,999)
        local ping=0
        pcall(function()
            local net=Stats:FindFirstChild("Network")
            local ss=net and net:FindFirstChild("ServerStatsItem")
            local dp=ss and ss:FindFirstChild("Data Ping")
            if dp then ping=tonumber(string.match(dp:GetValueString(),"%d+%.?%d*")) or 0
            else        ping=(LocalPlayer:GetNetworkPing() or 0)*1000 end
        end)
        metricsLbl.Text=string.format("High  |   %d FPS  |  %d ms",fps,math.max(0,math.floor(ping+0.5)))
        frames,elapsed=0,0
    end
    if bGrad and bGrad.Parent then bGrad.Rotation=(tick()*85)%360 end
end)
